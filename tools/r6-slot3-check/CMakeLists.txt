cmake_minimum_required(VERSION 3.20)
project(R6Slot3Check LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(R6Slot3Check Slot3Check.cpp)
target_compile_features(R6Slot3Check PRIVATE cxx_std_20)
target_include_directories(R6Slot3Check PRIVATE ../../modules/wxl-modern-render/src)
enable_testing()
add_test(NAME R6Slot3PolicyTransaction COMMAND R6Slot3Check)
add_executable(R6ProductionCheck ProductionCheck.cpp)
target_compile_features(R6ProductionCheck PRIVATE cxx_std_20)
target_include_directories(R6ProductionCheck PRIVATE ../../modules/wxl-modern-render/src)
add_test(NAME R6ProductionTransaction COMMAND R6ProductionCheck)
if(WIN32)
    add_executable(R6ShaderCompileCheck ShaderCompileCheck.cpp)
    target_compile_features(R6ShaderCompileCheck PRIVATE cxx_std_20)
    target_compile_definitions(R6ShaderCompileCheck PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_include_directories(R6ShaderCompileCheck PRIVATE ../../modules/wxl-modern-render/src)
    target_link_libraries(R6ShaderCompileCheck PRIVATE d3dcompiler)
    add_test(NAME R6SM3ShaderCompile COMMAND R6ShaderCompileCheck)
endif()
add_executable(R6NativeBridgeCheck NativeBridgeCheck.cpp)
target_compile_features(R6NativeBridgeCheck PRIVATE cxx_std_20)
target_include_directories(R6NativeBridgeCheck PRIVATE ../../modules/wxl-modern-render/src)
add_test(NAME R6NativeBridgeAndLifecycle COMMAND R6NativeBridgeCheck)
