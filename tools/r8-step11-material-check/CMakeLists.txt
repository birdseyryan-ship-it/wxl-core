cmake_minimum_required(VERSION 3.20)
project(C28ProofChecks LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(C28CoreCheck C28CoreCheck.cpp)
target_include_directories(C28CoreCheck PRIVATE ../../src)
enable_testing()
add_test(NAME C28CoreCheck COMMAND C28CoreCheck)
if(WIN32)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "Runtime ABI check requires Win32")
  endif()
  add_executable(C28RuntimeCheck C28RuntimeCheck.cpp)
  target_include_directories(C28RuntimeCheck PRIVATE ../../src ../../modules/wxl-modern-render/src)
  target_compile_definitions(C28RuntimeCheck PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
  add_test(NAME C28RuntimeCheck COMMAND C28RuntimeCheck "${CMAKE_CURRENT_SOURCE_DIR}/fixtures")
endif()
